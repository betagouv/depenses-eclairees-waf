# Authorize any method (put, patch, delete, ...) on /rest/ endpoint
SecRule REQUEST_URI "@beginsWith /rest/" \
    "id:2001,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveById=911100"

# n8n editor lists credential types via /types/credentials.json; CRS sees a restricted file
SecRule REQUEST_FILENAME "@streq /types/credentials.json" \
    "id:2002,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveTargetById=930130;REQUEST_FILENAME"

# n8n's /rest/ API is authenticated and its bodies are user-authored
# content (workflow JS, SQL, node params, credential refs), so CRS
# injection rules always fire: 930120 on ".credentials" arg names (JSON
# keys become arg names), 932* on jsCode, etc. Exempt the RCE, SQLi, PHP
# and LFI tags on ARGS and ARGS_NAMES for the whole API. Auth and
# SSO/OAuth routes, api-keys, test-only routes (e2e, debug) and the
# instance-ai gateway stay inspected; /rest/login is also the
# rate-limited login endpoint. The exclusion list tracks n8n's
# unauthenticated routes and must be rechecked on n8n upgrades; see
# "Updating the n8n /rest/ exemption" in AGENTS.md.
SecRule REQUEST_URI "@rx ^/rest/(?!login|api-keys|forgot-password|change-password|resolve-signup-token|resolve-password-token|resolve-change-email-token|confirm-email-change|change-email|owner/setup|invitations|sso|auth|oauth1-credential|oauth2-credential|e2e|debug|instance-ai/gateway)" \
    "id:2004,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-RCE;ARGS,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-SQLI;ARGS,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-PHP;ARGS,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-LFI;ARGS,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-RCE;ARGS_NAMES,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-SQLI;ARGS_NAMES,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-PHP;ARGS_NAMES,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-LFI;ARGS_NAMES"
