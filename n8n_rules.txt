# Authorize any method (put, patch, delete, ...) on /rest/ endpoint
SecRule REQUEST_URI "@beginsWith /rest/" \
    "id:2001,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveById=911100"

# n8n editor lists credential types via /types/credentials.json; CRS sees a restricted file
SecRule REQUEST_FILENAME "@streq /types/credentials.json" \
    "id:2002,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveTargetById=930130;REQUEST_FILENAME"

# Credential arg names trip rule 930120 on ".credentials" from
# lfi-os-files.data; drop ARGS_NAMES from 930120 on this route only.
SecRule REQUEST_URI "@beginsWith /rest/credentials/" \
    "id:2003,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveTargetById=930120;ARGS_NAMES"

# Saving a workflow ships each Code node's JavaScript in jsCode; the
# "function" keyword in it then trips CRS RCE rules (ex: 932235), and
# credential references (json.nodes.array_3.credentials.gristApi.id)
# trip the LFI rule 930120 on ".credentials" from lfi-os-files.data.
# Whole workflow bodies are user-authored code and data, so exempt the
# RCE, SQLi, PHP and LFI tags on ARGS and ARGS_NAMES for this route only.
# ARGS_NAMES is needed because JSON keys become argument names, and the
# node index varies with every workflow so no single rule target covers
# them.
SecRule REQUEST_URI "@beginsWith /rest/workflows" \
    "id:2004,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-RCE;ARGS,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-SQLI;ARGS,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-PHP;ARGS,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-LFI;ARGS,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-LFI;ARGS_NAMES"
