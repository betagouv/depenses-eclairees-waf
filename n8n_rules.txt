# Authorize any method (put, patch, delete, ...) on /rest/ endpoint
SecRule REQUEST_URI "@beginsWith /rest/" \
    "id:2001,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveById=911100"

# n8n editor lists credential types via /types/credentials.json; CRS sees a restricted file
SecRule REQUEST_FILENAME "@streq /types/credentials.json" \
    "id:2002,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveTargetById=930130;REQUEST_FILENAME"

# Credential arg names trip rule 930120 on ".credentials" from
# lfi-os-files.data; drop ARGS_NAMES from 930120 on this route only.
SecRule REQUEST_URI "@beginsWith /rest/credentials/" \
    "id:2003,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveTargetById=930120;ARGS_NAMES"
