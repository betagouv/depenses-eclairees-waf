# Authorize any method (put, delete, ...) on /api/ endpoint
SecRule REQUEST_URI "@beginsWith /api/" \
    "id:1001,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveById=911100"


# Edit question
SecRule REQUEST_URI "@beginsWith /api/dataset" \
    "id:1002,\
    phase:1,\
    t:none,\
    pass,\
    nolog,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-SQLI;ARGS:json.stages.array_0.native,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-RCE;ARGS:json.stages.array_0.native,\
    ctl:ruleRemoveTargetByTag=OWASP_CRS/ATTACK-PHP;ARGS:json.stages.array_0.native"


# Download answer

# Save question: the JSON body holds user-authored SQL (dataset_query)
# and Metabase-generated type strings in result_metadata (base_type,
# effective_type, field_ref.base-type) such as "type/*" that libinjection
# flags. Exempt every arg from the SQLi, RCE and PHP checks; other
# endpoints, ARGS_NAMES and the other CRS groups stay inspected.
